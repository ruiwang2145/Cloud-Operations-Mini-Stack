# Runtime dependencies for the Cloud Operations Mini Stack.

# Pinned exactly. An unpinned requirements file means the image that runs in
# production is not the image that passed CI, and "it worked yesterday" becomes
# an unfalsifiable claim. Bump deliberately, one package at a time, with the test
# suite as the safety net.

djangorestframework==3.18.1

# --- WSGI server ------------------------------------------------------------

# --- Configuration ----------------------------------------------------------
python-dotenv==1.2.3
